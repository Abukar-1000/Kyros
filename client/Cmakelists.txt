# Minimum version of CMake required
cmake_minimum_required(VERSION 3.28)

# Project name and version
project(Kyros VERSION 1.0)

# Specify the C++ standard
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED True)

set(
    SOURCES
    main.cpp
)

set(
    PROCESS_ITTERATOR_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/include/ProcessItterator/ProcessItterator.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/ProcessItterator/ProcessItterator.cpp
)

set(
    PROCESS_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/include/Process/Process.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Process/Process.cpp
)

# Add an executable called 'Kyros' from source 'main.cpp'
add_executable(
    Kyros 
    ${SOURCES}
    ${PROCESS_SOURCES}
    ${PROCESS_ITTERATOR_SOURCES}
)

if(MSVC)
    target_compile_options(Kyros PRIVATE /std:c++latest)
endif()

target_include_directories(
    Kyros
    PRIVATE
    ${PROCESS_SOURCES}
    ${PROCESS_ITTERATOR_SOURCES}
    ${CMAKE_CURRENT_SOURCE_DIR}/dependancies
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Header
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Header
)
