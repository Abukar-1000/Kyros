# Minimum version of CMake required
cmake_minimum_required(VERSION 3.28)

# Project name and version
project(Kyros VERSION 1.0)

# Specify the C++ standard

set(CMAKE_CXX_STANDARD_REQUIRED True)

set(
    SOURCES
    main.cpp
)

set(
    PROCESS_ITTERATOR_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/include/Process/Itterator/ProcessItterator.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Process/Itterator/ProcessItterator.cpp
)

set(
    PROCESS_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/include/Process/Process.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Process/Process.cpp 
    ${CMAKE_CURRENT_SOURCE_DIR}/include/Process/Metadata/ProcessMetadata.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Process/Metadata/ProcessMetadata.cpp
)

set(
    PROCESS_OBSERVER_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/include/Process/Context/ProcessContext.h
    ${CMAKE_CURRENT_SOURCE_DIR}/include/Process/Observer/ProcessObserver.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Process/Observer/ProcessObserver.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Process/Context/ProcessContext.cpp
)

set(
    INTER_PROCESS_COMMUNICATION_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/include/State/InterProcessCom/Pipe/Pipe.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/State/InterProcessCom/Pipe/Pipe.cpp

    ${CMAKE_CURRENT_SOURCE_DIR}/include/State/InterProcessCom/PipeBuilder/PipeBuilder.h

    ${CMAKE_CURRENT_SOURCE_DIR}/src/Command/Context/CommandContextSingleton.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/include/Command/Process/DeleteCommand.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Command/Process/DeleteCommand.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/include/Command/Context/CommandContext.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Command/Context/CommandContext.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/include/Command/Visitors/ProcessVisitor.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Command/Visitors/ProcessVisitor.cpp
)

set(
    BGWORKER_SOURCES
    
    ${CMAKE_CURRENT_SOURCE_DIR}/include/BGWorker/IBGWorker.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/BGWorker/IBGWorker.cpp
    
    ${CMAKE_CURRENT_SOURCE_DIR}/include/BGWorker/Command/CommandWorker.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/BGWorker/Command/CommandWorker.cpp
    
    ${CMAKE_CURRENT_SOURCE_DIR}/include/BGWorker/Context/BGWorkerParams.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/BGWorker/Context/BGWorkerParams.cpp

    ${CMAKE_CURRENT_SOURCE_DIR}/src/BGWorker/Context/BGWorkerContext.cpp
    
    ${CMAKE_CURRENT_SOURCE_DIR}/include/BGWorker/Context/BGWorkerContextSingleton.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/BGWorker/Context/BGWorkerContextSingleton.cpp

    
    ${CMAKE_CURRENT_SOURCE_DIR}/include/BGWorker/Gateway/GatewayWorker.h

    MODELS_SOURCES

    ${CMAKE_CURRENT_SOURCE_DIR}/models/Object/Object.h

# Add an executable called 'Kyros' from source 'main.cpp'
add_executable(
    Kyros 
    ${SOURCES}
    ${MODELS_SOURCES}
    ${PROCESS_SOURCES}
    ${PROCESS_ITTERATOR_SOURCES}
    ${PROCESS_OBSERVER_SOURCES}
    ${INTER_PROCESS_COMMUNICATION_SOURCES}
    ${COMMAND_SOURCES}
    ${BGWORKER_SOURCES}
    ${VISITORS_SOURCES}
)

if(MSVC)
    target_compile_options(Kyros PRIVATE /std:c++latest)
endif()

target_include_directories(
    Kyros
    PRIVATE
    ${MODELS_SOURCES}
    ${PROCESS_SOURCES}
    ${PROCESS_ITTERATOR_SOURCES}
    ${PROCESS_OBSERVER_SOURCES}
    ${INTER_PROCESS_COMMUNICATION_SOURCES}
    ${COMMAND_SOURCES}
    ${BGWORKER_SOURCES}
    ${VISITORS_SOURCES}
    ${CMAKE_CURRENT_SOURCE_DIR}/dependancies
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Header
    ${CMAKE_CURRENT_SOURCE_DIR}/src/Header
